Bug 1 — The widget never starts polling when an agent manually takes over
The widget only sets isHandoffMode = true when the /api/chat response returns data.source === 'handoff'. But when an agent clicks "Take Over" manually from the dashboard, no chat request is made — so the widget never finds out. It stays in AI mode, never polls, never sees the agent's reply.
Bug 2 — The sync route returns ALL messages but the widget APPENDS them
Your sync route returns the full message array. But the updated widget does:
js
messages = [...messages, ...data.messages]; // APPENDS
If sync returns 4 old messages and widget already has 4, it becomes 8 — duplicates everywhere. The old widget replaced (messages = data.messages), the new one appends. They're out of sync.
Bug 3 — /api/chat doesn't handle sessions already in human mode
When a customer sends another message while an agent has taken over, your chat route probably still hits the AI. It should detect mode === 'human' and return that mode to the widget so polling starts.
The complete fix — 3 files:
Fix 1 — /api/chat/sync/route.ts — slice correctly + add after support:
ts
import { NextRequest, NextResponse } from 'next/server'
import { supabase } from '@/lib/supabase'
import { Redis } from '@upstash/redis'
import { getCorsHeaders, handleOptions } from '@/lib/cors'

const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
})

export const dynamic = 'force-dynamic'

export async function OPTIONS(req: NextRequest) {
  const origin = req.headers.get('origin')
  const contextId = req.nextUrl.searchParams.get('ctx') ?? ''
  const cors = await getCorsHeaders(contextId, origin)
  return handleOptions(cors)
}

export async function GET(req: NextRequest) {
  const origin = req.headers.get('origin')
  const contextId = req.nextUrl.searchParams.get('ctx') ?? ''
  const sessionId = req.nextUrl.searchParams.get('session_id')
  // Widget tells us how many messages it already has
  const after = parseInt(req.nextUrl.searchParams.get('after') || '0')
  const cors = await getCorsHeaders(contextId, origin)

  if (!contextId || !sessionId) {
    return NextResponse.json(
      { error: 'Missing ctx or session_id', messages: [], mode: 'ai' },
      { status: 400, headers: cors }
    )
  }

  try {
    // ── FAST PATH: Check Redis first (sub-5ms) ─────────────────────────
    const cachedMode = await redis.get(`session_mode:${sessionId}`)

    // If not in handoff — return immediately, no Supabase needed
    if (!cachedMode || cachedMode === 'ai') {
      return NextResponse.json(
        { messages: [], mode: 'ai', total: after },
        { headers: cors }
      )
    }

    // ── SLOW PATH: Only runs for active handoff sessions ───────────────
    const { data: allMessages, error } = await supabase
      .from('messages')
      .select('role, content, source_url, created_at, agent_id')
      .eq('session_id', sessionId)
      .eq('context_id', contextId)
      .order('created_at', { ascending: true })

    if (error) {
      return NextResponse.json(
        { error: error.message, messages: [], mode: String(cachedMode) },
        { status: 500, headers: cors }
      )
    }

    const total = allMessages?.length || 0

    // Only send messages the widget doesn't already have
    const newMessages = (allMessages || []).slice(after)

    // Re-verify mode from DB in case Redis is stale
    const { data: sessionData } = await supabase
      .from('sessions')
      .select('mode')
      .eq('session_id', sessionId)
      .single()

    const mode = sessionData?.mode || String(cachedMode)

    // If mode changed back to ai, sync Redis
    if (mode === 'ai') {
      await redis.set(`session_mode:${sessionId}`, 'ai', { ex: 3600 })
    }

    return NextResponse.json(
      { messages: newMessages, mode, total },
      { headers: cors }
    )

  } catch (error: any) {
    return NextResponse.json(
      { error: error.message, messages: [], mode: 'ai' },
      { status: 500, headers: cors }
    )
  }
}

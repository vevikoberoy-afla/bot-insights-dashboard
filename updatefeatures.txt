/**
 * Chatwoot API Client
 * Connects to self-hosted Chatwoot instance on Oracle Cloud VM.
 * All calls are server-side only. Never import this in client components.
 */

import type {
  ChatwootContact,
  ChatwootConversation,
  ChatwootMessage,
  ChatwootInbox,
  CreateContactParams,
  CreateConversationParams,
} from '@/types/chatwoot'

const CHATWOOT_URL = process.env.CHATWOOT_URL
const TOKEN = process.env.CHATWOOT_API_TOKEN
const ACCOUNT = process.env.CHATWOOT_ACCOUNT_ID

if (!CHATWOOT_URL || !TOKEN || !ACCOUNT) {
  console.warn('[Chatwoot] Missing env vars — integration disabled')
}

/** Base fetch wrapper with auth headers */
async function chatwootFetch<T>(
  path: string,
  options: RequestInit = {}
): Promise<T> {
  const url = `${CHATWOOT_URL}/api/v1/accounts/${ACCOUNT}${path}`
  const res = await fetch(url, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      'api_access_token': TOKEN!,
      ...options.headers,
    },
  })
  if (!res.ok) {
    const error = await res.text()
    throw new Error(`Chatwoot API error ${res.status}: ${error}`)
  }
  return res.json() as Promise<T>
}

/** Create a new inbox for a bot */
export async function createInbox(botName: string): Promise<ChatwootInbox> {
  return chatwootFetch<ChatwootInbox>('/inboxes', {
    method: 'POST',
    body: JSON.stringify({
      name: botName,
      channel: {
        type: 'api',
        webhook_url: `${process.env.NEXTAUTH_URL}/api/webhooks/chatwoot`,
      },
    }),
  })
}

/** Find or create a contact by phone number */
export async function findOrCreateContact(
  params: CreateContactParams
): Promise<ChatwootContact> {
  // Search first to avoid duplicates
  if (params.phone_number) {
    try {
      const search = await chatwootFetch<{ payload: ChatwootContact[] }>(
        `/contacts/search?q=${encodeURIComponent(params.phone_number)}&page=1`
      )
      if (search.payload && search.payload.length > 0) {
        return search.payload[0]
      }
    } catch {
      // Contact not found — create new one below
    }
  }
  return chatwootFetch<ChatwootContact>('/contacts', {
    method: 'POST',
    body: JSON.stringify(params),
  })
}

/** Create a new conversation linked to a contact and inbox */
export async function createConversation(
  params: CreateConversationParams
): Promise<ChatwootConversation> {
  return chatwootFetch<ChatwootConversation>('/conversations', {
    method: 'POST',
    body: JSON.stringify(params),
  })
}

/** Send a message to an existing conversation */
export async function sendMessage(
  conversationId: number,
  content: string,
  type: 'incoming' | 'outgoing',
  isPrivate = false
): Promise<ChatwootMessage> {
  return chatwootFetch<ChatwootMessage>(
    `/conversations/${conversationId}/messages`,
    {
      method: 'POST',
      body: JSON.stringify({
        content,
        message_type: type,
        private: isPrivate,
      }),
    }
  )
}

/** Toggle conversation status to open (triggers agent alert) */
export async function triggerHandoff(
  conversationId: number
): Promise<ChatwootConversation> {
  return chatwootFetch<ChatwootConversation>(
    `/conversations/${conversationId}/toggle_status`,
    {
      method: 'POST',
      body: JSON.stringify({ status: 'open' }),
    }
  )
}

/** Add a label to a conversation */
export async function addLabel(
  conversationId: number,
  label: string
): Promise<void> {
  await chatwootFetch(`/conversations/${conversationId}/labels`, {
    method: 'POST',
    body: JSON.stringify({ labels: [label] }),
  })
}

/** Get all messages for a conversation */
export async function getMessages(
  conversationId: number
): Promise<ChatwootMessage[]> {
  const res = await chatwootFetch<{ payload: ChatwootMessage[] }>(
    `/conversations/${conversationId}/messages`
  )
  return res.payload ?? []
}

/** Resolve a conversation when session ends */
export async function resolveConversation(
  conversationId: number
): Promise<void> {
  await chatwootFetch(`/conversations/${conversationId}/toggle_status`, {
    method: 'POST',
    body: JSON.stringify({ status: 'resolved' }),
  })
}

/**
 * Dittofeed Integration
 * Uses the official @dittofeed/sdk-node package (MIT licensed).
 * Syncs contacts and tracks events for campaign targeting.
 * All calls are server-side only.
 */

import DittofeedSdk from '@dittofeed/sdk-node'
import type {
  DittofeedIdentifyParams,
  DittofeedTrackParams,
} from '@/types/dittofeed'

let sdk: InstanceType<typeof DittofeedSdk> | null = null

function getSDK(): InstanceType<typeof DittofeedSdk> {
  if (!sdk) {
    if (!process.env.DITTOFEED_URL || !process.env.DITTOFEED_WRITE_KEY) {
      throw new Error('[Dittofeed] Missing DITTOFEED_URL or DITTOFEED_WRITE_KEY')
    }
    sdk = new DittofeedSdk({
      host: process.env.DITTOFEED_URL,
      writeKey: process.env.DITTOFEED_WRITE_KEY,
    })
  }
  return sdk
}

/**
 * Sync a contact to Dittofeed CRM.
 * Call this whenever a new contact is created in Supabase.
 * This makes the contact available for campaign segmentation.
 */
export async function identifyContact(
  params: DittofeedIdentifyParams
): Promise<void> {
  try {
    const df = getSDK()
    await df.identify({
      userId: params.userId,
      traits: params.traits,
    })
    await df.flush()
  } catch (error) {
    // Non-fatal — log and continue. Dittofeed failure should not
    // break the main chat flow.
    console.error('[Dittofeed] identify failed:', error)
  }
}

/**
 * Track a behavioral event for a contact.
 * Used to trigger journey automations and segment updates.
 *
 * Example events:
 * - 'chat_started'
 * - 'asked_about_pricing'
 * - 'handoff_requested'
 * - 'campaign_replied'
 */
export async function trackEvent(
  params: DittofeedTrackParams
): Promise<void> {
  try {
    const df = getSDK()
    await df.track({
      userId: params.userId,
      event: params.event,
      properties: params.properties,
    })
    await df.flush()
  } catch (error) {
    console.error('[Dittofeed] track failed:', error)
  }
}

  const mode = req.nextUrl.searchParams.get('hub.mode')
  const token = req.nextUrl.searchParams.get('hub.verify_token')

  return new Response('Forbidden', { status: 403 })

FILE: app/api/whatsapp/incoming/route.ts
PURPOSE: Central router for ALL inbound WhatsApp messages.
         Decides: send to AI bot OR forward to Chatwoot human agent.
         This is the single webhook URL you register in Meta dashboard.
----------------------------------------------------------------

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { sendMessage as chatwootSendMessage } from '@/lib/chatwoot'
import { trackEvent } from '@/lib/dittofeed'

export async function POST(req: NextRequest) {
  try {
    const body = await req.json()

    // Meta sends a nested payload structure
    const entry = body?.entry?.[0]
    const change = entry?.changes?.[0]
    const value = change?.value

    // Ignore status updates (delivered, read receipts) — only handle messages
    if (!value?.messages || value.messages.length === 0) {
      return Response.json({ status: 'ignored' })
    }

    const message = value.messages[0]
    const from = message.from                // customer's WhatsApp number
    const messageText = message.text?.body   // message content
    const phoneNumberId = value.metadata?.phone_number_id

    if (!messageText || !from) {
      return Response.json({ status: 'ignored' })
    }

    const supabase = createClient()

    // 1. Find the bot linked to this WhatsApp number
    const { data: bot } = await supabase
      .from('bots')
      .select('id, client_id, chatwoot_inbox_id, whatsapp_phone_number_id')
      .eq('whatsapp_phone_number_id', phoneNumberId)
      .single()

    if (!bot) {
      console.error('[WhatsApp] No bot found for phone_number_id:', phoneNumberId)
      return Response.json({ status: 'no_bot_found' })
    }

    // 2. Find or create contact in Supabase
    let { data: contact } = await supabase
      .from('contacts')
      .select('*')
      .eq('phone', from)
      .eq('client_id', bot.client_id)
      .single()

    if (!contact) {
      const { data: newContact } = await supabase
        .from('contacts')
        .insert({
          client_id: bot.client_id,
          phone: from,
          channel: 'whatsapp',
        })
        .select()
        .single()
      contact = newContact
    }

    // 3. Find active session for this contact
    const { data: session } = await supabase
      .from('sessions')
      .select('*')
      .eq('bot_id', bot.id)
      .in('mode', ['ai', 'waiting', 'human'])
      .order('created_at', { ascending: false })
      .limit(1)
      .single()

    // 4. Route based on session mode
    if (session?.mode === 'human' && session?.chatwoot_conversation_id) {
      // Active human handoff — forward message to Chatwoot agent
      await chatwootSendMessage(
        session.chatwoot_conversation_id,
        messageText,
        'incoming'
      )

      // Save to our messages table too
      await supabase.from('messages').insert({
        session_id: session.id,
        content: messageText,
        sender_type: 'visitor',
        role: 'user',
      })

      return Response.json({ status: 'forwarded_to_human' })
    }

    // 5. AI handles the conversation
    // Create new session if none exists
    let activeSessionId = session?.id
    if (!session) {
      const { data: newSession } = await supabase
        .from('sessions')
        .insert({
          bot_id: bot.id,
          mode: 'ai',
          channel: 'whatsapp',
        })
        .select()
        .single()
      activeSessionId = newSession?.id

      // Track new conversation in Dittofeed
      if (contact && activeSessionId) {
        await trackEvent({
          userId: contact.id,
          event: 'whatsapp_conversation_started',
          properties: { botId: bot.id, channel: 'whatsapp' },
        })
      }
    }

    // 6. Call your existing AI chat endpoint internally
    const aiResponse = await fetch(
      `${process.env.NEXTAUTH_URL}/api/chat`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          message: messageText,
          sessionId: activeSessionId,
          botId: bot.id,
          channel: 'whatsapp',
        }),
      }
    )

    const aiData = await aiResponse.json()

    // 7. Send AI reply back via Meta WhatsApp API
    await fetch(
      `https://graph.facebook.com/v18.0/${phoneNumberId}/messages`,
      {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${process.env.WHATSAPP_ACCESS_TOKEN}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          messaging_product: 'whatsapp',
          to: from,
          type: 'text',
          text: { body: aiData.reply },
        }),
      }
    )

    return Response.json({ status: 'ai_handled' })

  } catch (error) {
    console.error('[WhatsApp Incoming] Error:', error)
    return Response.json({ error: 'Internal error' }, { status: 500 })
  }
}

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import {
  findOrCreateContact,
  createConversation,
} from '@/lib/chatwoot'
import { identifyContact } from '@/lib/dittofeed'

export async function POST(req: NextRequest) {
  try {
    const { botId, visitorName, visitorEmail } = await req.json()

    if (!botId) {
      return Response.json({ error: 'botId is required' }, { status: 400 })
    }

    const supabase = createClient()

    // 1. Get bot details including Chatwoot inbox ID
    const { data: bot, error: botError } = await supabase
      .from('bots')
      .select('id, client_id, chatwoot_inbox_id')
      .eq('id', botId)
      .single()

    if (botError || !bot) {
      return Response.json({ error: 'Bot not found' }, { status: 404 })
    }

    // 2. Create a new session in Supabase
    const { data: session, error: sessionError } = await supabase
      .from('sessions')
      .insert({
        bot_id: botId,
        mode: 'ai',
        channel: 'widget',
      })
      .select()
      .single()

    if (sessionError || !session) {
      return Response.json({ error: 'Session creation failed' }, { status: 500 })
    }

    // 3. Create/find contact in Supabase
    const { data: contact } = await supabase
      .from('contacts')
      .insert({
        client_id: bot.client_id,
        name: visitorName ?? 'Website Visitor',
        email: visitorEmail ?? null,
        channel: 'widget',
      })
      .select()
      .single()

    // 4. Create contact + conversation in Chatwoot (if inbox exists)
    if (bot.chatwoot_inbox_id) {
      try {
        const chatwootContact = await findOrCreateContact({
          name: visitorName ?? 'Website Visitor',
          email: visitorEmail,
        })

        const chatwootConversation = await createConversation({
          inbox_id: bot.chatwoot_inbox_id,
          contact_id: chatwootContact.id,
          status: 'pending',
        })

        // Save Chatwoot IDs back to session
        await supabase
          .from('sessions')
          .update({
            chatwoot_conversation_id: chatwootConversation.id,
            chatwoot_contact_id: chatwootContact.id,
          })
          .eq('id', session.id)

        // Sync contact to Dittofeed for campaign targeting
        if (contact) {
          await identifyContact({
            userId: contact.id,
            traits: {
              name: visitorName ?? 'Website Visitor',
              email: visitorEmail,
              channel: 'widget',
              clientId: bot.client_id,
              createdAt: new Date().toISOString(),
            },
          })
        }
      } catch (chatwootError) {
        // Non-fatal — AI chat still works without Chatwoot
        console.error('[Chat Init] Chatwoot setup failed:', chatwootError)
      }
    }

    return Response.json({
      sessionId: session.id,

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { triggerHandoff, addLabel } from '@/lib/chatwoot'
import { trackEvent } from '@/lib/dittofeed'

export async function POST(req: NextRequest) {
  try {
    const { sessionId } = await req.json()

    if (!sessionId) {
      return Response.json({ error: 'sessionId required' }, { status: 400 })
    }

    const supabase = createClient()

    // 1. Get session with Chatwoot conversation ID
    const { data: session, error } = await supabase
      .from('sessions')
      .select('*, contacts(*)')
      .eq('id', sessionId)
      .single()

    if (error || !session) {
      return Response.json({ error: 'Session not found' }, { status: 404 })
    }

    // 2. Update session mode to 'waiting'
    await supabase
      .from('sessions')
      .update({ mode: 'waiting' })
      .eq('id', sessionId)

    // 3. Alert Chatwoot agents
    if (session.chatwoot_conversation_id) {
      try {
        await triggerHandoff(session.chatwoot_conversation_id)
        await addLabel(session.chatwoot_conversation_id, 'needs-human')
      } catch (chatwootError) {
        console.error('[Handoff] Chatwoot alert failed:', chatwootError)
      }
    }

    // 4. Track in Dittofeed
    if (session.contacts?.id) {
      await trackEvent({
        userId: session.contacts.id,
        event: 'handoff_requested',
        properties: { sessionId, channel: session.channel },
      })
    }

    return Response.json({ status: 'waiting', message: 'Human agent alerted' })

  } catch (error) {
    console.error('[Handoff] Error:', error)
    return Response.json({ error: 'Internal error' }, { status: 500 })
  }
}

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'

export async function GET(req: NextRequest) {
  try {
    const sessionId = req.nextUrl.searchParams.get('sessionId')
    const since = req.nextUrl.searchParams.get('since') // ISO timestamp

    if (!sessionId) {
      return Response.json({ error: 'sessionId required' }, { status: 400 })
    }

    const supabase = createClient()

    // Get current session mode
    const { data: session } = await supabase
      .from('sessions')
      .select('mode')
      .eq('id', sessionId)
      .single()

    if (!session) {
      return Response.json({ error: 'Session not found' }, { status: 404 })
    }

    // If AI is handling — no need to poll, return empty
    if (session.mode === 'ai') {
      return Response.json({ mode: 'ai', messages: [] })
    }

    // If human is handling — return new human messages since last poll
    if (session.mode === 'human' || session.mode === 'waiting') {
      let query = supabase
        .from('messages')
        .select('*')
        .eq('session_id', sessionId)
        .eq('sender_type', 'human')
        .order('created_at', { ascending: true })

      if (since) {
        query = query.gt('created_at', since)
      }

      const { data: messages } = await query

      return Response.json({
        mode: session.mode,
        messages: messages ?? [],
      })
    }

    if (session.mode === 'resolved') {
      return Response.json({ mode: 'resolved', messages: [] })
    }

    return Response.json({ mode: session.mode, messages: [] })

  } catch (error) {
    console.error('[Sync] Error:', error)
    return Response.json({ error: 'Internal error' }, { status: 500 })
  }
}

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import type { ChatwootWebhookPayload } from '@/types/chatwoot'

export async function POST(req: NextRequest) {
  try {
    const payload = await req.json() as ChatwootWebhookPayload

    // Verify webhook secret
    const signature = req.headers.get('x-chatwoot-signature')
    const secret = process.env.CHATWOOT_WEBHOOK_SECRET
    if (secret && signature !== secret) {
      return Response.json({ error: 'Unauthorized' }, { status: 401 })
    }

    // Only handle agent-sent outgoing messages
    // Ignore: bot messages, activity logs, system messages
    if (
      payload.event !== 'message_created' ||
      payload.message_type !== 'outgoing' ||
      payload.sender?.type !== 'agent' ||
      !payload.content ||
      !payload.conversation?.id
    ) {
      return Response.json({ status: 'ignored' })
    }

    const supabase = createClient()

    // Find our session by Chatwoot conversation ID
    const { data: session } = await supabase
      .from('sessions')
      .select('id, mode, channel')
      .eq('chatwoot_conversation_id', payload.conversation.id)
      .single()

    if (!session) {
      console.warn('[Chatwoot Webhook] No session for conversation:', payload.conversation.id)
      return Response.json({ status: 'session_not_found' })
    }

    // Save human message to our messages table
    await supabase.from('messages').insert({
      session_id: session.id,
      content: payload.content,
      sender_type: 'human',
      role: 'assistant',
      chatwoot_message_id: payload.id,
    })

    // Update session mode to human (in case it was 'waiting')
    if (session.mode === 'waiting') {
      await supabase
        .from('sessions')
        .update({
          mode: 'human',
          last_human_message_at: new Date().toISOString(),
        })
        .eq('id', session.id)
    }

    // If this is a WhatsApp session — forward human reply to customer via Meta API
    if (session.channel === 'whatsapp') {
      // Get contact phone from session
      const { data: sessionData } = await supabase
        .from('sessions')
        .select('*, contacts(*)')
        .eq('id', session.id)
        .single()

      const customerPhone = sessionData?.contacts?.phone
      const bot = await supabase
        .from('bots')
        .select('whatsapp_phone_number_id')
        .eq('id', sessionData?.bot_id)
        .single()

      if (customerPhone && bot.data?.whatsapp_phone_number_id) {
        await fetch(
          `https://graph.facebook.com/v18.0/${bot.data.whatsapp_phone_number_id}/messages`,
          {
            method: 'POST',
            headers: {
              'Authorization': `Bearer ${process.env.WHATSAPP_ACCESS_TOKEN}`,
              'Content-Type': 'application/json',
            },
            body: JSON.stringify({
              messaging_product: 'whatsapp',
              to: customerPhone,
              type: 'text',
              text: { body: payload.content },
            }),
          }
        )
      }
    }

    return Response.json({ status: 'saved' })

  } catch (error) {
    console.error('[Chatwoot Webhook] Error:', error)
    return Response.json({ error: 'Internal error' }, { status: 500 })
  }
}

async function umamiFetch<T>(path: string, options: RequestInit = {}): Promise<T> {
  const res = await fetch(`${UMAMI_URL}/api${path}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      'x-umami-api-key': API_KEY!,
      ...options.headers,
    },
  })
  if (!res.ok) {
    const error = await res.text()
    throw new Error(`Umami API error ${res.status}: ${error}`)
  }
  return res.json() as Promise<T>
}

  console.warn('[Peppermint] Missing env vars — ticketing integration disabled')

async function peppermintFetch<T>(path: string, options: RequestInit = {}): Promise<T> {
  const res = await fetch(`${PEPPERMINT_URL}/api/v1${path}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${API_KEY}`,
      ...options.headers,
    },
  })
  if (!res.ok) {
    const error = await res.text()
    throw new Error(`Peppermint API error ${res.status}: ${error}`)
  }
  return res.json() as Promise<T>
}

import { NextRequest } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import type { PeppermintWebhookPayload } from '@/types/peppermint'

export async function POST(req: NextRequest) {
  try {
    const payload = await req.json() as PeppermintWebhookPayload

    const signature = req.headers.get('x-peppermint-signature')
    const secret = process.env.PEPPERMINT_WEBHOOK_SECRET
    if (secret && signature !== secret) {
      return Response.json({ error: 'Unauthorized' }, { status: 401 })
    }

    if (!payload.ticket) {
      return Response.json({ status: 'ignored' })
    }

    const supabase = createClient()

    const update: Record<string, unknown> = {
      status: payload.ticket.status,
      assignee: payload.ticket.assignedTo ?? null,
    }
    if (payload.ticket.status === 'resolved' || payload.ticket.status === 'closed') {
      update.resolved_at = new Date().toISOString()
    }

    await supabase
      .from('tickets')
      .update(update)
      .eq('peppermint_ticket_id', payload.ticket.id)

    return Response.json({ status: 'synced' })
  } catch (error) {
    console.error('[Peppermint Webhook] Error:', error)
    return Response.json({ error: 'Internal error' }, { status: 500 })
  }
}

/** get_segment — list/count contacts matching filter criteria */
export async function getSegment(
  ctx: ToolContext,
  criteria: { topic?: string; sinceDays?: number; channel?: string }
): Promise<ToolResult<{ count: number; contactIds: string[] }>> {
  const supabase = createClient()
  let query = supabase.from('contacts').select('id').eq('client_id', ctx.clientId)
  if (criteria.channel) query = query.eq('channel', criteria.channel)
  if (criteria.sinceDays) {
    const since = new Date(Date.now() - criteria.sinceDays * 86400000).toISOString()
    query = query.gte('created_at', since)
  }
  const { data, error } = await query
  if (error) return { ok: false, error: error.message }
  return { ok: true, data: { count: data.length, contactIds: data.map(d => d.id) } }
}

FLOW 5: ANALYTICS
1. New client onboarded → /api/clients/create-website provisions a
   Umami website entry, saves umami_website_id on the client row.

3. Client opens their Analytics tab → page calls /api/analytics/traffic
   → server pulls real stats from Umami → renders in your own chart,
   fully white-labeled.
4. You (platform operator) open /admin/analytics → see the full Umami
   dashboard embedded directly, cross-client.

FLOW 6: TICKETING
1. AI handoff triggers (same as Flow 2 in Part 1).
2. /api/chat/handoff now also calls /api/tickets/create → opens a
   Peppermint ticket, saves it to the tickets table linked to the
   session.
3. Agent works the ticket in Peppermint (or the embedded Tickets tab).
4. Peppermint fires a webhook on status change → /api/webhooks/peppermint
   → tickets table updated → visible in your dashboard without leaving
   the product.

FLOW 7 (FUTURE): AGENT-DRIVEN CAMPAIGN
1. Business owner types a request into a future chat-based orchestrator.
2. Orchestrator calls getSegment() and getAnalyticsSummary() (read-only,
   no approval needed) to understand the audience and context.
3. Orchestrator calls proposeCampaign() → row created in agent_actions
   with status 'proposed'.
4. Owner sees the proposal in their dashboard, clicks Approve.
5. /api/agent-actions/[id]/decide flips status to 'approved', logs to
   audit_logs with actor_type = 'human'.
6. executeApprovedAction() runs, delegates to the existing Dittofeed
   integration from Part 1, flips status to 'executed'.
